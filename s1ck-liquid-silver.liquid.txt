<script>
(function(){
  var root = document.getElementById('s1ck-{{ section.id }}');
  if(!root) return;

  var selected   = root.querySelector('[data-s1ck-selected]');
  var priceEl    = root.querySelector('[data-s1ck-price]');
  var compareEl  = root.querySelector('[data-s1ck-compare]');
  var stickyPrice= root.querySelector('[data-s1ck-sticky-price]');
  var qtyInput   = root.querySelector('[data-qty-input]');
  var addBtn     = root.querySelector('[data-s1ck-add]');
  var sticky     = root.querySelector('[data-s1ck-sticky]');

  /* qty stepper */
  var qtyWrap = root.querySelector('[data-s1ck-qty]');
  if(qtyWrap){
    qtyWrap.querySelector('[data-dec]').onclick=function(){qtyInput.value=Math.max(1,(parseInt(qtyInput.value)||1)-1)};
    qtyWrap.querySelector('[data-inc]').onclick=function(){qtyInput.value=(parseInt(qtyInput.value)||1)+1};
  }

  /* variant selector */
  root.querySelectorAll('[data-s1ck-variants] .size').forEach(function(el){
    el.addEventListener('click',function(){
      if(el.hasAttribute('data-soldout')) return;
      root.querySelectorAll('[data-s1ck-variants] .size').forEach(function(x){x.classList.remove('active')});
      el.classList.add('active');
      if(selected) selected.value = el.getAttribute('data-variant-id');
      if(priceEl) priceEl.textContent = el.getAttribute('data-price');
      if(stickyPrice) stickyPrice.textContent = el.getAttribute('data-price');
      var cmp = el.getAttribute('data-compare');
      if(compareEl){ compareEl.textContent = cmp || ''; compareEl.style.display = cmp ? '' : 'none'; }
    });
  });

  /* AJAX add to cart */
  function addToCart(id, qty, btn){
    if(!id){ alert('No variant configured.'); return; }
    var original = btn ? btn.textContent : '';
    if(btn){ btn.textContent = 'Adding…'; btn.disabled = true; }
    fetch('/cart/add.js',{
      method:'POST',
      headers:{'Content-Type':'application/json','Accept':'application/json'},
      body: JSON.stringify({ items:[{ id: parseInt(id), quantity: parseInt(qty)||1 }] })
    })
    .then(function(r){ if(!r.ok) return r.json().then(function(e){throw e}); return r.json(); })
    .then(function(){
      if(btn){ btn.textContent='Added ✓'; setTimeout(function(){btn.textContent=original;btn.disabled=false;},1400); }
      openCartDrawer();
    })
    .catch(function(err){
      if(btn){ btn.textContent=original; btn.disabled=false; }
      alert((err && err.description) ? err.description : 'Could not add to cart.');
    });
  }

  /* open the theme's cart drawer — covers Dawn + most common themes.
     If your drawer doesn't open, replace the body of this function
     with your theme's own trigger (or window.location='/cart'). */
  function openCartDrawer(){
    document.dispatchEvent(new CustomEvent('cart:refresh',{bubbles:true}));
    document.dispatchEvent(new CustomEvent('cart:build'));
    // Dawn / Horizon style
    var drawer = document.querySelector('cart-drawer, cart-notification');
    if(drawer){
      fetch('/?section_id=cart-drawer').then(function(r){return r.text()}).then(function(html){
        try{
          var el=new DOMParser().parseFromString(html,'text/html').querySelector('cart-drawer');
          if(el && drawer.tagName==='CART-DRAWER'){ drawer.innerHTML=el.innerHTML; }
        }catch(e){}
        if(drawer.classList) drawer.classList.remove('is-empty');
        if(typeof drawer.open==='function') drawer.open();
        else drawer.classList.add('active','animate','drawer--open');
        document.body.classList.add('overflow-hidden');
      }).catch(function(){});
      return;
    }
    // fallbacks used by other themes
    document.dispatchEvent(new CustomEvent('ajaxProduct:added'));
    document.body.classList.add('js-drawer-open','cart-open');
    var t=document.querySelector('[data-cart-drawer-toggle],[href="#cart-drawer"],.js-drawer-open-right');
    if(t) t.click();
  }

  if(addBtn) addBtn.addEventListener('click',function(){ addToCart(selected && selected.value, qtyInput && qtyInput.value, addBtn); });
  var addSticky = root.querySelector('[data-s1ck-add-sticky]');
  if(addSticky) addSticky.addEventListener('click',function(){ addToCart(selected && selected.value, qtyInput && qtyInput.value, addSticky); });

  /* bundle + addon buttons */
  root.querySelectorAll('[data-s1ck-bundle]').forEach(function(el){
    el.addEventListener('click',function(){ addToCart(el.getAttribute('data-variant-id'), el.getAttribute('data-qty'), el.tagName==='BUTTON'?el:null); });
  });

  /* sticky reveal */
  if(sticky){
    window.addEventListener('scroll',function(){
      if(window.scrollY>700) sticky.classList.add('show'); else sticky.classList.remove('show');
    },{passive:true});
  }

  /* single-open FAQ */
  root.querySelectorAll('details').forEach(function(d){
    d.addEventListener('toggle',function(){ if(d.open) root.querySelectorAll('details').forEach(function(o){ if(o!==d) o.open=false; }); });
  });
})();
</script>
